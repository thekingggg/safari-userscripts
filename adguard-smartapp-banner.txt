! AdGuard custom filter — Kill Smart App Banners & "Open in App" promos
! Works on Safari iOS/macOS with AdGuard. Place in a custom filter list.
! ---------------------------------------------------------------
! 1) HARD BLOCK: strip the HTTP response header that triggers the native iOS Smart App Banner.
!    This is the ONLY reliable way to stop Safari's blue "MỞ" banner.
! ---------------------------------------------------------------
||reddit.com^$removeheader=apple-itunes-app
||redd.it^$removeheader=apple-itunes-app
||tiktok.com^$removeheader=apple-itunes-app
||instagram.com^$removeheader=apple-itunes-app
||facebook.com^$removeheader=apple-itunes-app
||m.facebook.com^$removeheader=apple-itunes-app
||messenger.com^$removeheader=apple-itunes-app
||x.com^$removeheader=apple-itunes-app
||twitter.com^$removeheader=apple-itunes-app
||youtube.com^$removeheader=apple-itunes-app
||youtu.be^$removeheader=apple-itunes-app
||medium.com^$removeheader=apple-itunes-app
||pinterest.com^$removeheader=apple-itunes-app
||linkedin.com^$removeheader=apple-itunes-app
||zalo.me^$removeheader=apple-itunes-app
||shopee.vn^$removeheader=apple-itunes-app
||shopee.com^$removeheader=apple-itunes-app
||lazada.vn^$removeheader=apple-itunes-app
||zalo.me^$removeheader=apple-itunes-app
||telegram.org^$removeheader=apple-itunes-app
||telemetry.twitter.com^$removeheader=apple-itunes-app

! (Optional) Other header names some sites experiment with — harmless if absent
||*^$removeheader=x-ios-deeplink,domain=~localhost|~127.0.0.1
||*^$removeheader=x-app-banner,domain=~localhost|~127.0.0.1

! ---------------------------------------------------------------
! 2) COSMETIC: hide in‑page "open in app" / promo bars (useful on desktop or where header not used)
!    Note: cosmetic hiding cannot stop Safari's native banner; header stripping above is key.
! ---------------------------------------------------------------

##.smartbanner, ##.smart-app-banner, ##.app-banner, ##.app_banner, ##.appBanner, ##.app-download-banner, ##.open-in-app, ##.open-in-app-banner, ##.OpenInAppBanner, ##.app-promo, ##.appPromo, ##.pwa-install-banner, ##.mobile-app-banner, ##.appInstallPrompt
##[data-testid*="smartbanner"], ##[data-testid*="app-banner"], ##[id*="smartbanner"], ##[id*="app-banner"], ##[class*="SmartBanner"], ##[class*="smartbanner"], ##[class*="openInApp"], ##[class*="open-in-app"], ##[class*="AppInstall"], ##[class*="xpromo"]
